cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" RECALL_BASE_VERSION)
string(STRIP "${RECALL_BASE_VERSION}" RECALL_BASE_VERSION)
if(NOT RECALL_BASE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain major.minor.patch without leading zeroes")
endif()
project(ReCall VERSION ${RECALL_BASE_VERSION} LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
set(RECALL_BUILD_REVISION "0" CACHE STRING "MSIX/EXE revision (GitHub run number in CI)")
if(NOT RECALL_BUILD_REVISION MATCHES "^(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "RECALL_BUILD_REVISION must be a nonnegative integer without leading zeroes")
endif()
foreach(part IN ITEMS PROJECT_VERSION_MAJOR PROJECT_VERSION_MINOR PROJECT_VERSION_PATCH RECALL_BUILD_REVISION)
  if(${part} GREATER 65535)
    message(FATAL_ERROR "${part} must be at most 65535 (MSIX version limit)")
  endif()
endforeach()
set(RECALL_VERSION "${PROJECT_VERSION}.${RECALL_BUILD_REVISION}")
configure_file(assets/app.manifest.in generated/app.manifest @ONLY)
configure_file(assets/version.rc.in generated/version.rc @ONLY)
configure_file(src/version.h.in generated/version.h @ONLY)
message(STATUS "ReCall version: ${RECALL_VERSION}")
if(NOT WIN32)
  message(FATAL_ERROR "ReCall requires Windows 11 and the Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(sqlite URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip
  URL_HASH SHA256=1d3049dd0f830a025a53105fc79fd2ab9431aea99e137809d064d8ee8356b032)
FetchContent_MakeAvailable(sqlite)
add_library(sqlite3 STATIC "${sqlite_SOURCE_DIR}/sqlite3.c")
target_include_directories(sqlite3 PUBLIC "${sqlite_SOURCE_DIR}")
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
add_executable(ReCall WIN32 src/main.cpp src/config.cpp src/store.cpp src/clipboard.cpp
  src/catalog.cpp src/search.cpp src/app.cpp src/emoji_renderer.cpp src/theme.cpp src/appearance.cpp src/glass.cpp src/backdrop.cpp src/previews.cpp assets/resources.rc
  "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
target_include_directories(ReCall PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(ReCall PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
  _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(ReCall PRIVATE sqlite3 comctl32 dwmapi gdiplus ole32 shell32 shlwapi
  user32 gdi32 advapi32 windowsapp d2d1 dwrite CoreMessaging)
if(MSVC)
  target_compile_options(ReCall PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(ReCall PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest)
endif()
